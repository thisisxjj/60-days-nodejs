1: // DAY 04 - 练习 1：事件驱动的下载管理器

2: IMPORT { EVENTEMITTER } FROM 'NODE:EVENTS';

3: CLASS DOWNLOADMANAGER EXTENDS EVENTEMITTER {
4:   CONSTRUCTOR() {
5:     SUPER();
6:     THIS.QUEUE = [];
7:     THIS.RUNNING = 0;
8:     THIS.MAXCONCURRENT = 3;
9:   }

10:   DOWNLOAD(FILE) {
11:     THIS.QUEUE.PUSH(FILE);
12:     THIS._NEXT();
13:   }

14:   _NEXT() {
15:     WHILE (THIS.RUNNING < THIS.MAXCONCURRENT && THIS.QUEUE.LENGTH > 0) {
16:       CONST FILE = THIS.QUEUE.SHIFT();
17:       THIS.RUNNING++;
18:       THIS._SIMULATE(FILE);
19:     }
20:   }

21:   ASYNC _SIMULATE(FILE) {
22:     // 模拟文件大小 1MB ~ 5MB
23:     CONST TOTALSIZE = MATH.FLOOR(MATH.RANDOM() * 4 * 1024 * 1024) + 1024 * 1024;
24:     CONST CHUNKSIZE = MATH.FLOOR(TOTALSIZE / 20);
25:     LET DOWNLOADED = 0;

26:     TRY {
27:       WHILE (DOWNLOADED < TOTALSIZE) {
28:         AWAIT NEW PROMISE((R) => SETTIMEOUT(R, 100 + MATH.RANDOM() * 150));

29:         // 随机模拟 5% 概率出错
30:         IF (MATH.RANDOM() < 0.05) {
31:           THROW NEW ERROR('网络连接超时');
32:         }

33:         DOWNLOADED = MATH.MIN(DOWNLOADED + CHUNKSIZE, TOTALSIZE);
34:         CONST PERCENT = ((DOWNLOADED / TOTALSIZE) * 100).TOFIXED(1);
35:         THIS.EMIT('PROGRESS', { FILE, PERCENT: PARSEFLOAT(PERCENT), DOWNLOADED, TOTALSIZE });
36:       }

37:       THIS.EMIT('COMPLETE', { FILE, SIZE: TOTALSIZE });
38:     } CATCH (ERROR) {
39:       THIS.EMIT('ERROR', { FILE, ERROR });
40:     } FINALLY {
41:       THIS.RUNNING--;
42:       THIS._NEXT();
43:     }
44:   }
45: }

46: // ========== 演示 ==========

47: FUNCTION FORMATBYTES(BYTES) {
48:   RETURN (BYTES / 1024 / 1024).TOFIXED(2) + ' MB';
49: }

50: FUNCTION RENDERBAR(PERCENT, WIDTH = 20) {
51:   CONST FILLED = MATH.ROUND((PERCENT / 100) * WIDTH);
52:   RETURN '[' + '█'.REPEAT(FILLED) + '░'.REPEAT(WIDTH - FILLED) + ']';
53: }

54: CONST MANAGER = NEW DOWNLOADMANAGER();
55: CONST PROGRESS = {};

56: MANAGER.ON('PROGRESS', ({ FILE, PERCENT }) => {
57:   PROGRESS[FILE] = PERCENT;
58:   PROCESS.STDOUT.WRITE(`\R  ${FILE.PADEND(12)} ${RENDERBAR(PERCENT)} ${PERCENT.TOFIXED(1)}%   `);
59: });

60: MANAGER.ON('COMPLETE', ({ FILE, SIZE }) => {
61:   CONSOLE.LOG(`\N✅ ${FILE} 下载完成 — ${FORMATBYTES(SIZE)}`);
62: });

63: MANAGER.ON('ERROR', ({ FILE, ERROR }) => {
64:   CONSOLE.LOG(`\N❌ ${FILE} 下载失败 — ${ERROR.MESSAGE}`);
65: });

66: CONST FILES = ['FILE1.ZIP', 'FILE2.ZIP', 'FILE3.ZIP', 'FILE4.ZIP', 'FILE5.ZIP'];

67: CONSOLE.LOG(`开始下载 ${FILES.LENGTH} 个文件（最大并发: 3）\N`);
68: FOR (CONST F OF FILES) {
69:   MANAGER.DOWNLOAD(F);
70: }
